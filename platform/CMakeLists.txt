  message(FATAL_ERROR "Use a separate native build directory for B8_SANITIZE")

if(EMSCRIPTEN)
  # Runtime uses C++ exceptions for hardware-reset unwinding and model faults.
  add_compile_options(-fexceptions)
  add_link_options(-fexceptions)
endif()

  if(MSVC)
    message(FATAL_ERROR "This sanitizer preset is validated for GCC/Clang only")

  add_link_options(-fsanitize=address,undefined)

  message(FATAL_ERROR "B8_NUMERIC_PROFILE must be B8 or B16")
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  set(B8_POLICY_CLANG "${CMAKE_CXX_COMPILER}" CACHE FILEPATH "Clang driver for firmware policy checks")
else()
  find_program(B8_POLICY_CLANG NAMES clang++ clang++-20 clang++-19 clang++-18)
endif()

  message(FATAL_ERROR "LLVM clang++ is required to enforce firmware numeric widths; set B8_POLICY_CLANG")

set_property(CACHE B8_MEMORY_PROFILE PROPERTY STRINGS standard plus free)
get_filename_component(B8_COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
if(EMSCRIPTEN)
  execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -print-prog-name=llvm-size
    OUTPUT_VARIABLE B8_LLVM_SIZE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  set(B8_SIZE_TOOL "${B8_LLVM_SIZE}" CACHE FILEPATH "Firmware section measurement tool" FORCE)
else()
  find_program(B8_SIZE_TOOL NAMES llvm-size size HINTS "${B8_COMPILER_DIR}")
endif()
if(NOT B8_SIZE_TOOL)
  message(FATAL_ERROR "Set B8_SIZE_TOOL to llvm-size for firmware memory accounting")
endif()
add_custom_target(firmware_memory ALL
  COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_memory.py"
    --config "${CMAKE_CURRENT_SOURCE_DIR}/memory-profiles.json"
    --device "${B8_NUMERIC_PROFILE}" --tier "${B8_MEMORY_PROFILE}"
    --size-tool "${B8_SIZE_TOOL}" --output "${CMAKE_CURRENT_BINARY_DIR}/firmware-memory.json"
    --objects $<TARGET_OBJECTS:student_firmware>
  DEPENDS student_firmware COMMAND_EXPAND_LISTS VERBATIM)

  endif()
endforeach()
